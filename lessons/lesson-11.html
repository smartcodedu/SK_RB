local sign = script.Parent
local emergence_distance = 25
local originalText = ""
local newText = "Ты на верном пути"

local function checkPlayerDistance()
    local players = game.Players:GetPlayers()
    for _, player in ipairs(players) do
        local character = player.Character
        if not character then continue end
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local distance = (hrp.Position - sign.Position).Magnitude
        if distance <= emergence_distance then
            sign.SurfaceGui.TextLabel.Text = newText
        else
            sign.SurfaceGui.TextLabel.Text = originalText
        end
    end
end

while true do
    checkPlayerDistance()
    wait(0.1)
end

3 Отображение подсказки при приближении

Проверь, что скрипт Hint находится в объекте, табличка

📄 Скопируй в файл Hint 

local sign = script.Parent
local emergence_distance = 25
local originalText = ""
local newText = "Ты на верном пути"

local function checkPlayerDistance()
    local players = game.Players:GetPlayers()
    for _, player in ipairs(players) do
        local character = player.Character
        if not character then continue end
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local distance = (hrp.Position - sign.Position).Magnitude
        if distance <= emergence_distance then
            sign.SurfaceGui.TextLabel.Text = newText
        else
            sign.SurfaceGui.TextLabel.Text = originalText
        end
    end
end

while true do
    checkPlayerDistance()
    wait(0.1)
end

4 Урон огнём

В этот раз скрипт должен быть внутри модели костра, проверь

📄 Скопируй в файл Fire_Dmg 

local model = script.Parent
local DAMAGE = 15
local COOLDOWN = 1

local lastHit = {}

for _, part in model:GetDescendants() do
    if part:IsA("BasePart") then
        part.Touched:Connect(function(hit)
            local humanoid = hit.Parent:FindFirstChildOfClass("Humanoid")
            if not humanoid then return end
            local now = tick()
            if lastHit[humanoid] and now - lastHit[humanoid] < COOLDOWN then return end
            lastHit[humanoid] = now
            humanoid:TakeDamage(DAMAGE)
        end)
    end
end

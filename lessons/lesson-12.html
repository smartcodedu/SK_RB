1 Исчезновение двери при касании ключа

Проверь, что модели находится в Workspace

📄 Скопируй в файл DoorKey 

local key = workspace:WaitForChild("Key")
local door = workspace:WaitForChild("Door")

for _, part in key:GetDescendants() do
    if part:IsA("BasePart") then
        part.Touched:Connect(function(hit)
            local humanoid = hit.Parent:FindFirstChildOfClass("Humanoid")
            if humanoid then
                door:Destroy()
                key:Destroy()
                local msg = Instance.new("Message")
                msg.Parent = workspace
                msg.Text = "Дверь открыта"
                wait(2)
                msg:Destroy()
            end
        end)
    end
end

2 Сбор объектов

Внимательно следи за названием папки в Workspace, и за деталью Barrier

📄 Скопируй в файл Collecting 

local collection = workspace:WaitForChild("Collection")
local barrier = workspace:WaitForChild("Barrier")

local total = #collection:GetChildren()
local collected = 0

for _, object in collection:GetChildren() do
    for _, part in object:GetDescendants() do
        if part:IsA("BasePart") then
            part.Touched:Connect(function(hit)
                if not object.Parent then return end
                if hit.Parent:FindFirstChildOfClass("Humanoid") then
                    object:Destroy()
                    collected = collected + 1
                    if collected >= total then
                        barrier:Destroy()
                    end
                end
            end)
        end
    end
end

⭐ Подсветка ключа

Внимательно сравни свой и этот скрипт. Добавь несколько строк в свой(или удали свой и скопируй этот скрпт)

📄 Внеси изменения в скрипте для подсказки Hint 

local sign = script.Parent
local originalText = ""
local emergence_distance = 25
local newText = "Ты на верном пути"

local function checkPlayerDistance()
    local players = game.Players:GetPlayers()
    for _, player in ipairs(players) do
        local character = player.Character
        if not character then continue end
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local distance = (hrp.Position - sign.Position).Magnitude
        if distance <= emergence_distance then
            sign.SurfaceGui.TextLabel.Text = newText
            workspace.Key.Highlight.Enabled = true
        else
            sign.SurfaceGui.TextLabel.Text = originalText
        end
    end
end

while true do
    checkPlayerDistance()
    wait(0.1)
end

📄 Скопируй в файл DoubleJump 

local UIS = game:GetService("UserInputService")
local player = game.Players.LocalPlayer
local char = player.Character or player.CharacterAdded:Wait()
local humanoid = char:WaitForChild("Humanoid")

local jumps = 0
local maxJumps = 2
local canDoubleJump = false

humanoid.StateChanged:Connect(function(_, new)
    if new == Enum.HumanoidStateType.Landed then
        jumps = 0
        canDoubleJump = false
    elseif new == Enum.HumanoidStateType.Jumping then
        canDoubleJump = true
    end
end)

UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.Space and canDoubleJump and jumps < maxJumps then
        jumps += 1
        canDoubleJump = false
        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

🔒⭐ Ускорение при нажатии Shift

Материал закрыт! Введи пароль, который подсказал преподаватель.

Неверный пароль. Попробуй ещё.

Внимательно! Добавь LocalScript в папку StarterCharacterScripts
📄 Скопируй в файл Sprint 

local UIS = game:GetService("UserInputService")
local player = game.Players.LocalPlayer
local char = player.Character or player.CharacterAdded:Wait()
local humanoid = char:WaitForChild("Humanoid")

local normalSpeed = 16
local sprintSpeed = 32

UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.LeftShift then
        humanoid.WalkSpeed = sprintSpeed
    end
end)

UIS.InputEnded:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.LeftShift then
        humanoid.WalkSpeed = normalSpeed
    end
end)

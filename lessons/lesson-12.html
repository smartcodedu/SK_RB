1 Исчезновение двери при касании ключа

Проверь, что модели находится в Workspace

📄 Скопируй в файл DoorKey 

local key = workspace:WaitForChild("Key")
local door = workspace:WaitForChild("Door")

for _, part in key:GetDescendants() do
    if part:IsA("BasePart") then
        part.Touched:Connect(function(hit)
            local humanoid = hit.Parent:FindFirstChildOfClass("Humanoid")
            if humanoid then
                door:Destroy()
                key:Destroy()
                local msg = Instance.new("Message")
                msg.Parent = workspace
                msg.Text = "Дверь открыта"
                wait(2)
                msg:Destroy()
            end
        end)
    end
end

4 Урон огнём

В этот раз скрипт должен быть внутри модели костра, проверь

📄 Скопируй в файл Fire_Dmg 

local model = script.Parent
local DAMAGE = 15
local COOLDOWN = 1

local lastHit = {}

for _, part in model:GetDescendants() do
    if part:IsA("BasePart") then
        part.Touched:Connect(function(hit)
            local humanoid = hit.Parent:FindFirstChildOfClass("Humanoid")
            if not humanoid then return end
            local now = tick()
            if lastHit[humanoid] and now - lastHit[humanoid] < COOLDOWN then return end
            lastHit[humanoid] = now
            humanoid:TakeDamage(DAMAGE)
        end)
    end
end
